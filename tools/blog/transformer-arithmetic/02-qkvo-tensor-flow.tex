      \pgfmathtruncatemacro{\shade}{35+mod(13*\r+19*\c+7*\r*\c,45)}

          \fill[#5!\shade,rounded corners=.5pt] ({(\c-1)*.22+.012},{-(\r-1)*.22-.012}) rectangle ({\c*.22-.012},{-\r*.22+.012});

      \else
        \fill[#5!\shade,rounded corners=.5pt] ({(\c-1)*.22+.012},{-(\r-1)*.22-.012}) rectangle ({\c*.22-.012},{-\r*.22+.012});
      \fi
    }}
    \draw[black!55,thin] (0,0) rectangle (#4*.22,-#3*.22);
    \if\relax\detokenize{#6}\relax\else
      \node[font=\small,anchor=north] at (#4*.11,-#3*.22-.14) {$#6$};
      \node[font=\scriptsize,text=black!65,anchor=north] at (#4*.11,-#3*.22-.58) {$#7$};
    \fi
  \end{scope}
}
